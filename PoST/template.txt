(*Controller*)
PROGRAM Controller

VAR_INPUT
	Obj1 : BOOL;
	Obj2 : BOOL;
	Obj3 : BOOL;
	Obj4 : BOOL;
	Obj5 : BOOL;
	Drop_control : BOOL;
	Button1 : BOOL;
	Button2 : BOOL;
	Button3 : BOOL;
	Button4 : BOOL;
	Button5 : BOOL;
	Cancellation : BOOL;
	bank_terminal : UDINT;
	cash_terminal : UDINT;
END_VAR

VAR_OUTPUT
	Spring1: BOOL;
	Spring2: BOOL;
	Spring3: BOOL;
	Spring4: BOOL;
	Spring5: BOOL;
	text_display_idx : UINT;
	credit : UDINT;
END_VAR

VAR
	active_button : INT;
	price1: INT;
	price2: INT;
	price3: INT;
	price4: INT;
	price5: INT;
	cash_prev : UDINT;
	user_change : UDINT;
END_VAR

PROCESS Main_Loop
	
    STATE product_chosing
	    active_button := 0;
	    cash_prev := 0;
		(*Перенести  в Constqnt VAR*)
		price1 := 15;
		price2 := 10;
		price3 := 5;
		price4 := 7;
		price5 := 20;
		
		IF Cancellation THEN
			SET STATE give_change;
		END_IF
		
	    IF Button1 THEN
			active_button := 1;
			IF NOT Obj1 THEN
				text_display_idx := 1;
			END_IF
			IF Obj1 THEN
				SET STATE payment_wait;
			END_IF
			
	    END_IF
	    IF Button2 THEN
			active_button := 2;
			IF NOT Obj2 THEN
				text_display_idx := 1;
			END_IF
			IF Obj2 THEN
				SET STATE payment_wait;
			END_IF
	    END_IF
	    IF Button3 THEN
			active_button := 3;
			IF NOT Obj3 THEN
				text_display_idx := 1;
			END_IF
			IF Obj3 THEN
				SET STATE payment_wait;
			END_IF
	    END_IF
	    IF Button4 THEN
			active_button := 4;
			IF NOT Obj4 THEN
				text_display_idx := 1;
			END_IF
			IF Obj4 THEN
				SET STATE payment_wait;
			END_IF
	    END_IF
	    IF Button5 THEN
			active_button := 5;
			IF NOT Obj5 THEN
				text_display_idx := 1;
			END_IF
			IF Obj5 THEN
				SET STATE payment_wait;
			END_IF
	    END_IF
		TIMEOUT T#3s THEN
			text_display_idx := 0;
			RESTART; (*Reset timer or nothing*)
		END_TIMEOUT
    END_STATE
	(*При нажатии на кнопку надо таймер ресет*) 
	
	STATE payment_wait
		text_display_idx := 2;
		TIMEOUT T#3s THEN
			SET STATE insert_payment;
		END_TIMEOUT
	END_STATE

	STATE insert_payment
		text_display_idx := 3;
		
		IF Cancellation THEN
			SET STATE give_change;
		END_IF
		
		IF cash_terminal > 0 THEN
			IF cash_prev < cash_terminal THEN
				credit := credit + cash_terminal;
			END_IF
	

			IF active_button = 1 AND credit >= price1 THEN
				credit := credit - price1;
				SET STATE give_product;
			END_IF
			
			IF active_button = 2 AND credit >= price2 THEN
				credit := credit - price2;
				SET STATE give_product;
			END_IF
			
			IF active_button = 3 AND credit >= price3 THEN
				credit := credit - price3;
				SET STATE give_product;
			END_IF
			
			IF active_button = 4 AND credit >= price4 THEN
				credit := credit - price4;
				SET STATE give_product;
			END_IF
			
			IF active_button = 5 AND credit >= price5 THEN
				credit := credit - price5;
				SET STATE give_product;
			END_IF
		
		ELSE
				
			IF active_button = 1 AND bank_terminal >= price1 THEN
				credit := credit - price1;
				SET STATE give_product;
			END_IF
			
			IF active_button = 2 AND bank_terminal >= price2 THEN
				credit := credit - price2;
				SET STATE give_product;
			END_IF
			
			IF active_button = 3 AND bank_terminal >= price3 THEN
				credit := credit - price3;
				SET STATE give_product;
			END_IF
			
			IF active_button = 4 AND bank_terminal >= price4 THEN
				credit := credit - price4;
				SET STATE give_product;
			END_IF
			
			IF active_button = 5 AND bank_terminal >= price5 THEN
				credit := credit - price5;
				SET STATE give_product;
			END_IF
			
			(*IF bank_terminal > 0 THEN
				SET STATE give_product;
			END_IF*)
		END_IF
		
		cash_prev := cash_terminal;	
		
		TIMEOUT T#5s THEN
			SET STATE not_enough_money;
		END_TIMEOUT
		
	END_STATE
	
	STATE not_enough_money
		text_display_idx := 4;
		
		TIMEOUT T#1s THEN
			SET STATE insert_payment;
		END_TIMEOUT
	
	END_STATE

	STATE give_change
		IF credit > 0 THEN
			user_change := credit;
		END_IF
		credit := 0;

		TIMEOUT T#3s THEN
			SET STATE product_chosing;
		END_TIMEOUT
	
	END_STATE
	
	STATE give_product
		text_display_idx := 6;
		
		IF active_button = 1 THEN
			Spring1 := 1;
		END_IF
		IF active_button = 2 THEN
			Spring2 := 1;
		END_IF
		IF active_button = 3 THEN
			Spring3 := 1;
		END_IF
		IF active_button = 4 THEN
			Spring4 := 1;
		END_IF
		IF active_button = 5 THEN
			Spring5 := 1;
		END_IF
		
		IF Drop_control THEN
			Spring1 := 0;
			Spring2 := 0;
			Spring3 := 0;
			Spring4 := 0;
			Spring5 := 0;
			SET STATE give_change;
		END_IF
		
		TIMEOUT T#3s THEN
			Spring1 := 0;
			Spring2 := 0;
			Spring3 := 0;
			Spring4 := 0;
			Spring5 := 0;
			SET STATE give_product_timeout;
		END_TIMEOUT
		
	END_STATE
	
	STATE give_product_timeout
	
		TIMEOUT T#10s THEN
			SET STATE product_chosing;
		END_TIMEOUT
	
	END_STATE
	

END_PROCESS

END_PROGRAM
